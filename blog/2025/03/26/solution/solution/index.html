节流与防抖​

节流
function throttle(func, delay) {
  // 节流函数，在触发一次后，在delay毫秒内不再触发，直到delay毫秒后，才可再次触发
  let lastTime = 0
  let timeoutId = null

  return function (...args) {
    if (!timeoutId) {
      timeoutId = setTimeout(() => {
        timeoutId = null
        func.apply(this, args)
      }, delay)
    }
  }
}

// 测试代码
function onScroll() {
  console.log('Scroll event triggered:', Date.now())
}

const throttledOnScroll = throttle(onScroll, 1000)

// 模拟滚动事件
setInterval(throttledOnScroll, 200) // 每 200 毫秒触发一次滚动事件
防抖
function debounce(func, delay) {
  // 节流函数，在触发事件后，在给定的时间间隔内只执行一次函数
  let timeoutId
  let timeoutId

  return function (...args) {
    // 清除之前的定时器
    if (timeoutId) {
      clearTimeout(timeoutId)
    }

    // 设置一个新的定时器
    timeoutId = setTimeout(() => {
      func.apply(this, args)
    }, delay)
  }
}

// 测试代码
function onInput() {
  console.log('Input event triggered:', Date.now())
}

const debouncedOnInput = debounce(onInput, 1000)

// 模拟输入事件
setInterval(debouncedOnInput, 200) // 每 200 毫秒触发一次输入事件

函数柯里化​

测试代码
function addCurrying(fn) {
  const args = []

  function _curry(...newArgs) {
    args.push(...newArgs)
    if (args.length >= fn.length) {
      // fn.length 表示原函数的参数个数，args存储各个分级函数的参数
      return fn(...args)
    }
    return _curry
  }

  return _curry
}

// 示例
function add(a, b, c) {
  return a + b + c
}

const curriedAdd = addCurrying(add)

console.log(curriedAdd(1)(2)(3)) // 输出 6
console.log(curriedAdd(1, 2)(3)) // 输出 6
console.log(curriedAdd(1)(2, 3)) // 输出 6

class Cal {
  constructor(value) {
    this.value = value
  }

  add(num) {
    this.value += num
    return this // 返回对象自身，以支持链式调用
  }
  minus(num) {

  }
  subtract(num) {
    this.value -= num
    return this // 返回对象自身，以支持链式调用
  }

  multiply(num) {
    this.value *= num
    return this // 返回对象自身，以支持链式调用
  }

  }
}

const result = new Cal(10).add(5).subtract(3).multiply(2).get()

存储进位进行运算
function addLargeNumbers(num1, num2) {
  // 确保 num1 和 num2 都是字符串
  num1 = num1.toString()
  num2 = num2.toString()

  // 初始化结果数组和进位
  let result = []
  let carry = 0

  // 获取两个数字的长度
  let len1 = num1.length
  let len2 = num2.length

  // 计算最长的长度
  let maxLen = Math.max(len1, len2)

  // 从右到左逐位相加
  for (let i = 0; i < maxLen; i++) {
    // 获取当前位的数字，如果长度不足则补 0
    let digit1 = i < len1 ? parseInt(num1[len1 - 1 - i]) : 0
    let digit2 = i < len2 ? parseInt(num2[len2 - 1 - i]) : 0

    // 计算当前位的和
    let sum = digit1 + digit2 + carry

    // 计算新的进位
    carry = Math.floor(sum / 10)

    // 将当前位的结果存入结果数组
    result.push(sum % 10)
  }

  // 如果最后还有进位，则添加到结果数组中
  if (carry > 0) {
    result.push(carry)
  }

  // 将结果数组反转并转换为字符串
  return result.reverse().join('')
}

// 测试函数
const num1 = '123456789012345678901234567890'
const num2 = '987654321098765432109876543210'
const sum = addLargeNumbers(num1, num2)

console.log(sum) // 输出 "1111111110111111111011111111100"

JS 转换对象成为树结构​

利用map遍历数组成为树结构
const flatArray = [
  { id: 1, parentId: null, name: 'root1' },
  { id: 2, parentId: 1, name: 'child1' },
  { id: 3, parentId: 1, name: 'child2' },
  { id: 4, parentId: 2, name: 'grandchild1' },
  { id: 5, parentId: 3, name: 'grandchild2' }
]

function arrayToTree(items) {
  const map = new Map()
  const result = []

  // 创建一个 map，方便后续查找节点
  items.forEach(item => {
    map.set(item.id, { ...item, children: [] })
  })

  items.forEach(item => {
    const node = map.get(item.id)
    if (item.parentId === null) {
      // 如果是根节点，直接添加到 result
      result.push(node)
    } else {
      // 如果有父节点，找到父节点并将当前节点加入其 children 数组
      const parentNode = map.get(item.parentId)
      if (parentNode) {
        parentNode.children.push(node)
      }
    }
  })

  return result
}

const tree = arrayToTree(flatArray)
console.log(JSON.stringify(tree, null, 2))
利用递归实现
const flatArray = [
  { id: 1, parentId: null, name: 'root1' },
  { id: 2, parentId: 1, name: 'child1' },
  { id: 3, parentId: 1, name: 'child2' },
  { id: 4, parentId: 2, name: 'grandchild1' },
  { id: 5, parentId: 3, name: 'grandchild2' }
]

function arrayToTree(array, parentId = null) {
  return array
    .filter(item => item.parentId === parentId)
    .map(item => ({
      ...item,
      children: arrayToTree(array, item.id)
    }))
}

// 测试函数
const tree = arrayToTree(flatArray)
console.log(JSON.stringify(tree, null, 2)) // 输出结果 stringify的三个参数，第一个是json对象，第二个是 replacer，第三个是缩进空格数

解析 url​

利用URLSearchParams解析url
// 定义解析函数
function parseUrlParams(url) {
  // 创建 URL 对象
  const urlObj = new URL(url)

  // 创建 URLSearchParams 对象
  const params = new URLSearchParams(urlObj.search)

  // 初始化存储结果的对象
  const result = {}

  // 遍历 URLSearchParams 对象
  for (const [key, value] of params.entries()) {
    result[key] = value
  }

  return result
}

// 测试解析函数
const url = 'https://example.com?name=yupi&age=25&location=shanghai'
const parsedParams = parseUrlParams(url)

console.log(parsedParams)
// 输出结果如下
// {
//     name: 'yupi',
//     age: '25',
//     location: 'shanghai'
// }

双向数据绑定的实现​

defineProperty
<body>
  <div id="div1">
    <input type="text" v-model="name" />
    <br />
    姓名：{{name}}
  </div>

  <script>
    // 获取 input 元素和 div 容器
    const input = document.querySelector('input[v-model="name"]')
    const div = document.getElementById('div1')

    function updateView(value) {
      // 更新视图
      input.value = value
      div.innerHTML = div.innerHTML.replace(/\{\{name\}\}/g, value)
    }

    // 初始化模型对象
    const data = {}
    Object.defineProperty(data, 'name', {
      get() {
        return this._name
      },
      set(value) {
        // 设置值，参数为新值
        this._name = value
        updateView(value)
      }
    })

    // 设置初始值
    data.name = ''

    // 监听 input 输入变化
    input.addEventListener('input', e => {
      data.name = e.target.value
    })

    // 初始化视图
    updateView(data.name)
  </script>
</body>
proxy实现
<body>
  <div id="div1">
    <input type="text" v-model="name" />
    <br />
    姓名：{{name}}
  </div>

  <script>
    // 获取 input 元素和 div 容器
    const input = document.querySelector('input[v-model="name"]')
    const div = document.getElementById('div1')

    function updateView(value) {
      input.value = value
      div.innerHTML = div.innerHTML.replace(/\{\{name\}\}/g, value)
    }

    // 初始化模型对象并使用 Proxy 进行代理
    const data = new Proxy(
      {},
      {
        set(target, key, value) {
          // 设置值，参数为三个，第一个是目标对象，第二个是属性名，第三个是新值
          target[key] = value
          if (key === 'name') {
            updateView(value)
          }
          return true
        },
        get(target, key) {
          // 获取值，参数为两个，第一个是目标对象，第二个是属性名
          return target[key]
        }
      }
    )

    // 设置初始值
    data.name = ''

    // 监听 input 输入变化
    input.addEventListener('input', e => {
      data.name = e.target.value
    })

    // 初始化视图
    updateView(data.name)
  </script>
</body>
